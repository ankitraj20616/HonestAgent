```python
"""Shared SVG drawings for Saarthi screens: The Wheel, the journey Wheel, the Trust ring.

All colors come from CSS classes in base.css. This module never writes a color value.
Spec: core_docs/design/2026-10-11-wheel-ui-design.md (sections 6 and 7).
"""

from __future__ import annotations

import html
import math

STAGES = ("capture", "research", "brief", "run", "verify", "review", "ship", "remember")
BEAD_TONES = ("road", "ok", "bad", "info", "idle")

_VIEW = 520  # the big Wheel is drawn in a 520 x 520 box
_C = _VIEW / 2
_RIM = 190.0
_LABEL_R = 232.0
_BEAD_STEP = 9.0  # degrees between beads that share a spoke


def _e(value: object) -> str:
    return html.escape(str(value))


def stage_angle(stage: str) -> float:
    """SVG angle in degrees (0 = 3 o'clock, clockwise). Capture sits at 12 o'clock."""
    if stage not in STAGES:
        raise ValueError(f"unknown stage {stage!r}; use one of {STAGES}")
    return -90.0 + 45.0 * STAGES.index(stage)


def point(cx: float, cy: float, r: float, deg: float) -> tuple[float, float]:
    a = math.radians(deg)
    return round(cx + r * math.cos(a), 1), round(cy + r * math.sin(a), 1)


def spread(n: int, step: float = _BEAD_STEP) -> list[float]:
    """Centered angle offsets for n beads on one spoke."""
    return [round((i - (n - 1) / 2) * step, 2) for i in range(n)]


def trust_tone(total: float) -> str:
    if total >= 70:
        return "ok"
    if total >= 40:
        return "road"
    return "bad"


def _arc(cls: str, center: float, r: float, start_deg: float, length: float) -> str:
    """A circle stroke that only shows `length` units, starting at `start_deg`."""
    circ = 2 * math.pi * r
    return (
        f'<circle class="{cls}" cx="{center}" cy="{center}" r="{r}" '
        f'stroke-dasharray="{length:.1f} {circ - length:.1f}" '
        f'transform="rotate({start_deg:.1f} {center} {center})"/>'
    )


def _spokes(cls: str, center: float, r: float) -> list[str]:
    out = []
    for k in range(4):
        x1, y1 = point(center, center, r, -90 + 45 * k)
        x2, y2 = point(center, center, r, 90 + 45 * k)
        out.append(f'<line class="{cls}" x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}"/>')
    return out


def _beads(beads: list[dict]) -> list[str]:
    by_stage: dict[str, list[dict]] = {}
    for bead in beads:
        stage_angle(bead["stage"])
        if bead.get("tone", "idle") not in BEAD_TONES:
            raise ValueError(f"bead tone must be one of {BEAD_TONES}, got {bead.get('tone')!r}")
        by_stage.setdefault(bead["stage"], []).append(bead)

    out: list[str] = []
    for stage, group in by_stage.items():
        base = stage_angle(stage)
        placed = []
        for bead, offset in zip(group, spread(len(group))):
            deg = base + offset
            x, y = point(_C, _C, _RIM, deg)
            placed.append((bead, x, y, deg))
        links: dict[str, list[float]] = {}
        for bead, _x, _y, deg in placed:
            if bead.get("link"):
                links.setdefault(bead["link"], []).append(deg)
        for degs in links.values():
            if len(degs) > 1:
                start, end = min(degs), max(degs)
                out.append(_arc("b-link", _C, _RIM, start, 2 * math.pi * _RIM * (end - start) / 360))
        for bead, x, y, _deg in placed:
            tone = bead.get("tone", "idle")
            size = float(bead.get("r", 14))
            dim = f' opacity="{float(bead["dim"])}"' if "dim" in bead else ""
            if bead.get("halo"):
                out.append(f'<circle class="b-halo" cx="{x}" cy="{y}" r="{size + 9}"/>')
            out.append(f'<circle class="b-{tone}" cx="{x}" cy="{y}" r="{size}"{dim}/>')
            if bead.get("n"):
                out.append(
                    f'<text class="b-txt b-txt-{tone}" x="{x}" y="{y + 3.5}" text-anchor="middle"{dim}>'
                    f"{_e(bead['n'])}</text>"
                )
    return out


def wheel(data: dict) -> str:
    """The big Wheel. data keys: beads, counts, hub, lit, size, capture_plus."""
    size = int(data.get("size", _VIEW))
    c = _C
    out = [
        f'<svg class="wheel" width="{size}" height="{size}" viewBox="0 0 {_VIEW} {_VIEW}" '
        f'role="img" aria-label="The Wheel">',
        f'<circle class="w-ring" cx="{c}" cy="{c}" r="238"/>',
        f'<circle class="w-rim" cx="{c}" cy="{c}" r="{_RIM}"/>',
    ]
    segment = 2 * math.pi * _RIM / 8
    for stage in data.get("lit", []):
        out.append(_arc("w-lit", c, _RIM, stage_angle(stage) - 22.5, segment))
    out += _spokes("w-spoke", c, _RIM)
    out.append(f'<circle class="w-hubglow" cx="{c}" cy="{c}" r="96"/>')
    out.append(f'<circle class="w-hub" cx="{c}" cy="{c}" r="70"/>')
    hub = data.get("hub") or {}
    if hub:
        out.append(f'<text class="w-big" x="{c}" y="{c - 4}" text-anchor="middle">{_e(hub.get("big", ""))}</text>')
        out.append(f'<text class="w-sub" x="{c}" y="{c + 19}" text-anchor="middle">{_e(hub.get("sub", ""))}</text>')
    counts = data.get("counts", {})
    for stage in STAGES:
        x, y = point(c, c, _LABEL_R, stage_angle(stage))
        out.append(f'<text class="w-lbl" x="{x}" y="{y + 4}" text-anchor="middle">{stage.upper()}</text>')
        if stage in counts:
            out.append(f'<text class="w-n" x="{x}" y="{y + 18}" text-anchor="middle">{_e(counts[stage])}</text>')
    beads = data.get("beads", [])
    if data.get("capture_plus", True):
        n_capture = sum(1 for b in beads if b.get("stage") == "capture")
        offset = (n_capture - 1) / 2 * _BEAD_STEP + 11 if n_capture else 0
        x, y = point(c, c, _RIM, stage_angle("capture") + offset)
        out.append(f'<circle class="b-plus" cx="{x}" cy="{y}" r="12"/>')
        out.append(f'<text class="b-plus-t" x="{x}" y="{y + 4.5}" text-anchor="middle">+</text>')
    out += _beads(beads)
    out.append("</svg>")
    return "".join(out)


def journey(stage: str, size: int = 46, tone: str = "road", notches: tuple[float, ...] = ()) -> str:
    """Small Wheel for one task: the arc is filled up to the current stage."""
    angle = stage_angle(stage)
    if tone not in BEAD_TONES:
        raise ValueError(f"tone must be one of {BEAD_TONES}, got {tone!r}")
    index = STAGES.index(stage)
    c, r = 23, 19
    done = 2 * math.pi * r * index / 8
    out = [
        f'<svg class="journey" width="{size}" height="{size}" viewBox="0 0 46 46" '
        f'role="img" aria-label="Stage {index + 1} of 8">',
        f'<circle class="j-rim" cx="{c}" cy="{c}" r="{r}"/>',
    ]
    if done:
        out.append(_arc("j-done", c, r, -90, done))
    out += _spokes("j-spoke", c, r)
    for f in notches:
        if not 0 <= f <= 1:
            raise ValueError(f"notch must be between 0 and 1, got {f}")
        deg = -90 + 360 * f
        x1, y1 = point(c, c, 15, deg)
        x2, y2 = point(c, c, 22.5, deg)
        out.append(f'<line class="j-notch" x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}"/>')
    out.append(f'<circle class="j-hub" cx="{c}" cy="{c}" r="3.5"/>')
    x, y = point(c, c, r, angle)
    out.append(f'<circle class="j-now j-now-{tone}" cx="{x}" cy="{y}" r="5"/>')
    out.append("</svg>")
    return "".join(out)


def trust_ring(checks: float, criteria: float, second: float, size: int = 250, label: str = "TRUST SCORE") -> str:
    """Trust score ring. checks 0..50, criteria 0..35, second opinion 0..15 (spec section 7)."""
    for name, value, top in (("checks", checks, 50), ("criteria", criteria, 35), ("second", second, 15)):
        if not 0 <= value <= top:
            raise ValueError(f"{name} must be between 0 and {top}, got {value}")
    total = round(checks + criteria + second)

    if size >= 120:
        c, r = 125, 100
        circ = 2 * math.pi * r
        out = [
            f'<svg class="trust" width="{size}" height="{size}" viewBox="0 0 250 250" '
            f'role="img" aria-label="Trust score {total}">',
            f'<circle class="tr-bg" cx="{c}" cy="{c}" r="{r}"/>',
        ]
        start = -90.0
        for cls, value in (("tr-c", checks), ("tr-k", criteria), ("tr-s", second)):
            if value > 0:
                out.append(_arc(cls, c, r, start, circ * value / 100))
            start += 360 * value / 100
        out.append(f'<text class="tr-num" x="{c}" y="{c + 12}" text-anchor="middle">{total}</text>')
        out.append(f'<text class="tr-lbl" x="{c}" y="{c + 36}" text-anchor="middle">{_e(label)}</text>')
        out.append("</svg>")
        return "".join(out)

    tone = trust_tone(total)
    if size >= 40:
        c, r, view, text = 26, 21, 52, True
    else:
        c, r, view, text = 10, 7, 20, False
    circ = 2 * math.pi * r
    out = [
        f'<svg class="trust-sm" width="{size}" height="{size}" viewBox="0 0 {view} {view}" '
        f'role="img" aria-label="Trust score {total}">',
        f'<circle class="tr-bg{"-mini" if not text else ""}" cx="{c}" cy="{c}" r="{r}"/>',
        _arc(f"tr-{tone}", c, r, -90, circ * total / 100),
    ]
    if text:
        out.append(f'<text class="tr-num-sm" x="{c}" y="{c + 5}" text-anchor="middle">{total}</text>')
    out.append("</svg>")
    return "".join(out)
```

Note: the mini ring background class is `tr-bg-mini` (thin stroke). The arc class is `tr-ok`, `tr-road` or `tr-bad`.

```python
"""Build the Saarthi milestone screens ("The Wheel" look, dark only) from one source.

    python3 core_docs/design/milestones/_src/build.py                    # all milestones + index.html
    python3 core_docs/design/milestones/_src/build.py M4_checks_proof_card  # one milestone only

Input:  _src/screens/<milestone folder>/<NN_slug>.html   (one fragment per screen)
Output: <milestone folder>/<NN_slug>.html and index.html (gallery of all screens)
Spec:   core_docs/design/2026-10-11-wheel-ui-design.md

A fragment starts with one JSON comment (front matter), then plain HTML:

    <!-- {"title": "Home", "note": "The Wheel. F1.3, F16.2", "shell": "app", "nav": "home", "dock": "full"} -->
    <div class="wheel-wrap"> ... </div>

shell = "app"  -> top bar + agents dock/rail are added. Optional keys:
    nav           home | board | memory | playbooks | settings   (old names: deck = home, room = board)
    dock          "full" (Home only) or "rail" (default)
    project       name shown in the project picker ("" shows "Add a project")
    needs         Needs You count (default 0)
    cost          today's cost (default "$4.12")
    agents        [{"tag": "cl|cx", "task": "T-0012", "title": "Reset API", "state": "needs you",
                    "tone": "road|bad|ok|muted", "line": "build · 18m", "cost": "$0.62"}, ...]
    slots         agent slots, a number (old text like "2 of 4" also works); default 4
    daemon        {"dot": "green", "pulse": false, "label": "saarthid running", "sub": "127.0.0.1:7878 · v0.9",
                   "note": "..."}   (label/sub/note may hold HTML)
    after_topbar  HTML placed under the top bar (status bars)
shell = "bare" -> nothing added (spike windows, CLI terminals, first-run wizard)

Drawings, expanded at build time by wheel_svg.py:
    <x-wheel data='{"beads": [...], "counts": {...}, "hub": {...}, "lit": [...], "size": 520}'></x-wheel>
    <x-journey stage="run" tone="road" size="46" notches="0.2,0.6"></x-journey>

def _e(value: object) -> str:
    return html.escape(str(value))


def slot_count(value: object) -> int:
    numbers = re.findall(r"\d+", str(value))
    if not numbers:
        sys.exit(f"slots must hold a number, got {value!r}")
    return int(numbers[-1])


def parse(path: Path) -> tuple[dict, str]:
    raw = path.read_text()
    m = FRONT.match(raw)
    if not m:
        sys.exit(f"{path}: missing front matter comment")
    meta = json.loads(m.group(1))
    body = raw[m.end():].strip()
    css_bits = re.findall(r'style="([^"]*)"', body) + re.findall(r"<style>(.*?)</style>", body, re.S)
    for bit in css_bits:
        bad = COLOR_LITERAL.search(bit)
        if bad:
            sys.exit(f"{path}: color literal '{bad.group(0)}' in CSS '{bit[:60]}'. Use theme variables.")
    for key in ("title", "note", "shell"):
        if key not in meta:
            sys.exit(f"{path}: front matter needs '{key}'")
    return meta, body


def expand(body: str, where: str) -> str:
    """Replace <x-wheel>, <x-journey> and <x-trust> with SVG."""

    def attrs(text: str) -> dict:
        return dict(ATTR.findall(text))

    def do_wheel(m: re.Match) -> str:
        return wheel_svg.wheel(json.loads(m.group(1)))

    def do_journey(m: re.Match) -> str:
        a = attrs(m.group(1))
        notches = tuple(float(x) for x in a.get("notches", "").split(",") if x.strip())
        return wheel_svg.journey(a["stage"], int(a.get("size", 46)), a.get("tone", "road"), notches)

    def do_trust(m: re.Match) -> str:
        a = attrs(m.group(1))
        return wheel_svg.trust_ring(float(a.get("checks", 0)), float(a.get("criteria", 0)),
                                    float(a.get("second", 0)), int(a.get("size", 250)),
                                    a.get("label", "TRUST SCORE"))

    try:
        body = X_WHEEL.sub(do_wheel, body)
        body = X_JOURNEY.sub(do_journey, body)
        body = X_TRUST.sub(do_trust, body)
    except (ValueError, KeyError, json.JSONDecodeError) as err:
        sys.exit(f"{where}: bad drawing tag: {err}")
    return body


def _spark(seed: str) -> str:
    n = int(re.sub(r"\D", "", seed) or 7)
    points = []
    for i in range(11):
        n = (n * 1103515245 + 12345) % 2**31
        points.append(f"{i * 20},{3 + n % 13}")
    return " ".join(points)


def _agent_card(a: dict) -> str:
    tone = a.get("tone", "muted")
    return (
        f'<div class="ag-card{" wait" if tone == "road" else ""}">'
        f'<span class="agent {a["tag"]}">{a["tag"].upper()}</span>'

        f'<span class="ag-s tone-{tone}">{_e(a.get("state", ""))}</span>'
        f'<svg class="spark spark-{a["tag"]}" viewBox="0 0 200 18" preserveAspectRatio="none">'
        f'<polyline points="{_spark(a["task"])}"/></svg>'
        f'<div class="ag-m"><span>{_e(a.get("line", ""))}</span><span>{_e(a.get("cost", ""))}</span></div></div>'
    )


def _rail_item(a: dict) -> str:
    tone = a.get("tone", "muted")
    return (
        f'<div class="ra{" wait" if tone == "road" else ""}" title="{_e(a["task"])} · {_e(a.get("state", ""))}">'
        f'<span class="agent {a["tag"]}">{a["tag"].upper()}</span><i class="ra-dot tone-bg-{tone}"></i></div>'

def app_shell(meta: dict, body: str) -> str:
    nav = NAV_ALIASES.get(meta.get("nav", "home"), meta.get("nav", "home"))
    dock = meta.get("dock", "rail")
    project = meta.get("project", "acme-billing")
    needs = int(meta.get("needs", 0))
    agents = meta.get("agents", DEFAULT_AGENTS)
    free = max(0, slot_count(meta.get("slots", 4)) - len(agents))
    d = {**DEFAULT_DAEMON, **meta.get("daemon", {})}

    links = "".join(f'<a class="on">{label}</a>' if key == nav else f"<a>{label}</a>" for key, label in NAV)
    if needs:
        pill = f'<button class="needs-pill"><span class="dot road pulse"></span>{needs} need{"s" if needs == 1 else ""} you</button>'

def proto_nav(code: str, title: str, files: list[str], i: int) -> str:
    prev_link = f'<a href="{files[i - 1]}">&larr; prev</a>' if i > 0 else '<span class="faint">&larr; prev</span>'
    next_link = f'<a href="{files[i + 1]}">next &rarr;</a>' if i < len(files) - 1 else '<span class="faint">next &rarr;</span>'
    return (
        f'<div class="proto-nav"><span class="handle" title="Design navigator">{code} {i + 1}/{len(files)}</span>'
        f'<span class="more"><a href="../index.html">All screens</a><span class="sep">|</span>'
        f'<span>{code} · {i + 1}/{len(files)} · {_e(title)}</span><span class="sep">|</span>{prev_link}{next_link}</span></div>'
    )


def shared_css(folder: str) -> str:
    path = SCREENS / folder / "_shared.css"
    if not path.exists():
        return ""
    text = path.read_text()
    bad = COLOR_LITERAL.search(re.sub(r"/\*.*?\*/", "", text, flags=re.S))
    if bad:
        sys.exit(f"{path}: color literal '{bad.group(0)}'. Use theme variables.")
    return text


def css() -> str:
    return (SRC / "theme.css").read_text() + "\n" + (SRC / "base.css").read_text()


def page(meta: dict, body: str, code: str, nav_html: str, extra_css: str = "") -> str:
    inner = app_shell(meta, body) if meta["shell"] == "app" else body
    return f"""<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>Saarthi {code} · {_e(meta['title'])}</title>
<meta name="description" content="{_e(meta['note'])}">
{FONTS}
<style>
{css()}
{extra_css}
.after-top:empty {{ display: none; }}
</style>
</head>
<body>
{inner}
{nav_html}
</body>
</html>
"""


def gallery(entries: dict[str, list[tuple[str, dict]]]) -> str:
    sections = []
    for folder, (code, name) in MILESTONES.items():
        rows = entries.get(folder, [])
        if not rows:
            continue
        trs = "".join(
            f'<tr><td class="mono faint">{i + 1:02d}</td><td><a href="{folder}/{f}"><b>{_e(m["title"])}</b></a>'
            f'<div class="small muted">{_e(m["note"])}</div></td></tr>'
            for i, (f, m) in enumerate(rows)
        )
        sections.append(
            f'<section class="card"><div class="card-head"><h2>{code} · {name}</h2>'
            f'<span class="small muted">{len(rows)} screens · <span class="mono">{folder}/</span></span></div>'
            f'<table class="table"><tbody>{trs}</tbody></table></section>'
        )
    return f"""<!doctype html>
<html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1">
<title>Saarthi Milestone Screens</title>{FONTS}<style>{css()}</style></head>
<body><div class="page narrow">
<div class="page-head"><div class="row gap-12">{LOGO}<div>
<h1 class="title">Saarthi · UI by milestone</h1>
<div class="muted">"The Wheel" look, dark only. Built from <span class="mono">_src/</span>.
Spec: <span class="mono">core_docs/design/2026-10-11-wheel-ui-design.md</span>.</div></div></div></div>
{''.join(sections)}
</div></body></html>
"""


def main() -> None:
    entries: dict[str, list[tuple[str, dict]]] = {}
    total = 0
    only = sys.argv[1] if len(sys.argv) > 1 else None
    if only and only not in MILESTONES:
        sys.exit(f"unknown milestone folder {only!r}; use one of {list(MILESTONES)}")
    for folder, (code, _name) in MILESTONES.items():
        if only and folder != only:
            continue
        src_dir = SCREENS / folder
        if not src_dir.is_dir():
            continue
        files = sorted(p.name for p in src_dir.glob("*.html"))
        parsed = [parse(src_dir / f) for f in files]
        extra_css = shared_css(folder)
        entries[folder] = [(f, meta) for f, (meta, _b) in zip(files, parsed)]
        out = ROOT / folder
        out.mkdir(parents=True, exist_ok=True)
        for old_theme_dir in ("light", "dark"):
            if (out / old_theme_dir).is_dir():
                shutil.rmtree(out / old_theme_dir)
        for old in out.glob("*.html"):
            if old.name not in files:
                old.unlink()
        for i, (f, (meta, body)) in enumerate(zip(files, parsed)):
            body = expand(body, str(src_dir / f))
            (out / f).write_text(page(meta, body, code, proto_nav(code, meta["title"], files, i), extra_css))
            total += 1
    if not only:
        (ROOT / "index.html").write_text(gallery(entries))
    print(f"built {total} files in {len(entries)} milestones -> {ROOT}")


if __name__ == "__main__":
    main()
```

```python
"""Screenshot built screens with headless Chrome at 1440 x 900.

    python3 core_docs/design/milestones/_src/shoot.py OUT_DIR                     # every milestone
    python3 core_docs/design/milestones/_src/shoot.py OUT_DIR M3_desktop_ui_v1    # some milestones

PNG names: <milestone folder>__<screen>.png
"""

from __future__ import annotations

import subprocess
import sys
import tempfile
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
CHROME = "google-chrome"


def shoot(html_file: Path, png: Path) -> None:
    with tempfile.TemporaryDirectory() as profile:
        subprocess.run(
            [CHROME, "--headless=new", "--disable-gpu", "--no-sandbox", "--hide-scrollbars",
             f"--user-data-dir={profile}", "--window-size=1440,900", "--virtual-time-budget=4000",
             f"--screenshot={png}", html_file.as_uri()],
            capture_output=True, timeout=120, check=False,
        )


def main() -> None:
    if len(sys.argv) < 2:
        sys.exit(__doc__)
    out = Path(sys.argv[1]).resolve()
    out.mkdir(parents=True, exist_ok=True)
    folders = sys.argv[2:] or sorted(p.name for p in ROOT.iterdir() if p.is_dir() and p.name[:1] == "M")
    jobs = [(h, out / f"{f}__{h.stem}.png") for f in folders for h in sorted((ROOT / f).glob("*.html"))]
    with ThreadPoolExecutor(max_workers=8) as pool:
        list(pool.map(lambda job: shoot(*job), jobs))
    missing = [str(png) for _h, png in jobs if not png.exists()]
    print(f"{len(jobs) - len(missing)} of {len(jobs)} screenshots in {out}")
    if missing:
        sys.exit(f"missing: {missing[:5]}")


if __name__ == "__main__":
    main()
```

```markdown
- **F7.7 Trust score** (Must, M4; second-opinion part in M5)
  - One number from 0 to 100 that sums up the proof of a task:
    - Required checks: 50 points, by the share that pass.
    - Brief criteria: 35 points, by the share marked Met ("Not sure" counts 0 until confirmed).
    - Second opinion: 15 points if no high or medium finding is open, 8 if only medium findings are open, 0 if a high finding is open or no second opinion ran.
  - The daemon computes it. The UI only shows it (Proof ring, Home, Board cards).
  - It never blocks a move. The stage rules stay as in F2.5.
```

```markdown
### Decision N — Trust score is computed in the daemon from the Proof Card **(new)**
`checks/trust.py: trust_score(card: ProofCard, findings: list[ReviewFinding] | None) -> int` returns 0..100:
`round(50 * passed_required / required_total + 35 * met / criteria_total + second_part)`, where `second_part` is 0 in M4 (M5 fills it).
```
